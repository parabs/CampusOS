PNG to webp
only one folder 
Get-ChildItem -Filter *.png | ForEach-Object {
    cwebp -q 80 $_.FullName -o "$($_.DirectoryName)\$($_.BaseName).webp"
}

all subfolders 
Get-ChildItem -Recurse -Filter *.png | ForEach-Object {
    cwebp -q 80 $_.FullName -o "$($_.DirectoryName)\$($_.BaseName).webp"
}
